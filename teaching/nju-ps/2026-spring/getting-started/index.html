一个可以运行的例子

下面的组件引用同一目录下的 hello-vector.cpp。修改代码或输入后点击运行， 编译会在浏览器的 Web Worker 中完成。

hello-vector.cppGNU C++17
#include <algorithm>
#include <iostream>
#include <vector>

int main() {
    int n;
    if (!(std::cin >> n) || n < 0 || n > 1000) {
        std::cerr << "请输入 0 到 1000 之间的整数 n\n";
        return 1;
    }

    std::vector<int> values(n);
    for (int& value : values) {
        if (!(std::cin >> value)) {
            std::cerr << "输入的整数不够\n";
            return 1;
        }
    }

    std::sort(values.begin(), values.end());
    for (int value : values) std::cout << value << ' ';
    std::cout << '\n';
}
