C++ 运行试验

修改代码和输入，尝试一个小例子。编译和运行发生在你的浏览器中。

首次运行需要下载约 28 MB 编译器资源，之后复用同一个 Worker。 示例使用 GNU C++17；每次编译与运行合计最多 30 秒，可随时停止。

#include <algorithm>
#include <iostream>
#include <string>
#include <vector>

int main() {
    std::string name;
    int n;
    if (!(std::cin >> name >> n) || n < 0 || n > 1000) {
        std::cerr << "请输入名字和 0 到 1000 之间的元素个数\n";
        return 1;
    }
    std::vector<int> values(n);
    for (int& value : values) {
        if (!(std::cin >> value)) {
            std::cerr << "输入的整数不够\n";
            return 1;
        }
    }
    std::sort(values.begin(), values.end());
    std::cout << "Hello, " << name << "!\n";
    for (int value : values) std::cout << value << ' ';
    std::cout << '\n';
}
